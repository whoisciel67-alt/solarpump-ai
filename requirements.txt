import gradio as gr
import joblib
import pandas as pd
import requests

# Load your trained Random Forest model
model = joblib.load("model.joblib")


def fetch_open_meteo_weather(lat, lon):
  """Fetches live weather data from Open-Meteo API."""
  try:
    url = (
        f"https://api.open-meteo.com/v1/forecast?latitude={lat}&longitude={lon}"
        "&daily=shortwave_radiation_sum,temperature_2m_max,relative_humidity_2m_mean&timezone=auto"
    )
    response = requests.get(url, timeout=10)
    data = response.json()
    daily = data.get("daily", {})

    solar_rad = daily.get("shortwave_radiation_sum", [18.0])[0]
    max_temp = daily.get("temperature_2m_max", [30.0])[0]
    humidity = daily.get("relative_humidity_2m_mean", [65.0])[0]
    clear_sky = solar_rad * 1.2 if solar_rad else 22.0

    return float(solar_rad), float(clear_sky), float(max_temp), float(humidity)
  except Exception as e:
    print(f"Error fetching live weather: {e}")
    return 18.0, 22.0, 30.0, 65.0


def predict_solar_power(
    solar_radiation, clear_sky_radiation, max_temp, relative_humidity
):
  # Construct feature dataframe matching your training feature columns
  features = pd.DataFrame(
      [[solar_radiation, clear_sky_radiation, max_temp, relative_humidity]],
      columns=[
          "solar_radiation",
          "clear_sky_radiation",
          "max_temp",
          "relative_humidity",
      ],
  )

  # Generate model prediction
  prediction = model.predict(features)[0]

  # Determine recommendation tier
  if prediction >= 6.0:
    tier = "HIGH"
    recommendation = (
        "Run pumps during midday. Optimal solar energy conditions available."
    )
  elif prediction >= 3.5:
    tier = "MODERATE"
    recommendation = (
        "Run pumps during peak sunlight hours. Monitor battery levels."
    )
  else:
    tier = "LOW"
    recommendation = (
        "Conserve energy. Limit pumping to essential schedules or use backup"
        " power."
    )

  return round(prediction, 2), tier, recommendation


# Gradio Web Interface
def create_ui():
  with gr.Blocks(theme=gr.themes.Soft()) as demo:
    gr.Markdown("# ☀️ SolarPump AI Forecast & Irrigation Assistant")
    gr.Markdown(
        "Predict next-day solar energy generation and receive smart irrigation"
        " recommendations powered by your trained Random Forest model and"
        " Open-Meteo live weather."
    )

    with gr.Tab("Live Weather Integration"):
      with gr.Row():
        lat = gr.Number(label="Latitude", value=13.36, precision=4)
        lon = gr.Number(label="Longitude", value=103.85, precision=4)
      fetch_btn = gr.Button("Fetch Live Weather & Predict")

      with gr.Row():
        out_pred = gr.Number(
            label="Predicted Solar Energy Forecast (kWh/m²/day)"
        )
        out_tier = gr.Textbox(label="Pump Recommendation Tier")
      out_rec = gr.Textbox(label="Irrigation Guidance")

      fetch_btn.click(
          fn=lambda lat, lon: predict_solar_power(
              *fetch_open_meteo_weather(lat, lon)
          ),
          inputs=[lat, lon],
          outputs=[out_pred, out_tier, out_rec],
      )

    with gr.Tab("Manual Sliders (Testing)"):
      with gr.Row():
        slider_solar = gr.Slider(
            0, 30, value=18.0, step=0.1, label="Solar Radiation (MJ/m²)"
        )
        slider_clear = gr.Slider(
            0, 35, value=22.0, step=0.1, label="Clear-Sky Radiation (MJ/m²)"
        )
      with gr.Row():
        slider_temp = gr.Slider(
            10, 50, value=32.0, step=0.1, label="Maximum Temperature (°C)"
        )
        slider_humidity = gr.Slider(
            0, 100, value=65.0, step=1.0, label="Relative Humidity (%)"
        )

      calc_btn = gr.Button("Calculate Prediction")

      with gr.Row():
        manual_pred = gr.Number(
            label="Predicted Solar Energy Forecast (kWh/m²/day)"
        )
        manual_tier = gr.Textbox(label="Pump Recommendation Tier")
      manual_rec = gr.Textbox(label="Irrigation Guidance")

      calc_btn.click(
          fn=predict_solar_power,
          inputs=[slider_solar, slider_clear, slider_temp, slider_humidity],
          outputs=[manual_pred, manual_tier, manual_rec],
      )

    with gr.Tab("Model Performance"):
      gr.Markdown("""
      ### Model Evaluation Summary (2025 Holdout Evaluation)
      * **Random Forest MAE:** 1.865 (Best Performing Model)
      * **Linear Regression MAE:** 1.896
      * **Persistence MAE:** 1.966
      * **Tier Accuracy:** 92.3%
      """)

  return demo


if __name__ == "__main__":
  demo = create_ui()
  demo.launch(server_name="0.0.0.0", server_port=7860)
